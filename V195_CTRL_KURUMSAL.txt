     score >= thBuyScore and kgs >= thBuyKgs and leader >= thBuyLeader and rv >= thBuyRelVol and smc >= thBuySmc and _extended ? "WATCH" :

f_metrics(sym) =>

    rs = ta.roc(c, 10) - ta.roc(idx, 10)
    rsa = rs - rs[5]

    aboveMain = c > e50
    aboveSlow = c > e200
    health = stack and aboveMain ? 95 : c > e21 and e21 > e50 ? 72 : c > e50 ? 48 : 20
    htf = c > e200 and e21 > e50 ? 100 : c > e50 ? 65 : 35

    trendPts = stack and aboveMain ? 25 : c > e21 and e21 >= e50 ? 18 : c > e50 ? 10 : 0

    htfPts = aboveSlow and stack ? 12 : aboveMain ? 7 : 2

    loc = (c - l) / rng

    kgs = f_clamp(trendPts + emaPts + rsPts + volPts + secPts + flowPts + htfPts + math.round(smc * 0.12) - riskPenalty, 0, 100)

    fMomentum = f_clamp(50 + rs * 8 + rsa * 5, 0, 100)
    fQuality = stack and aboveMain ? 90.0 : c > e21 ? 60.0 : 25.0
    fVolStab = relVol > 2.5 ? 35.0 : relVol > 1.8 ? 55.0 : relVol > 0.8 ? 80.0 : 50.0

    domFactor = f_dominant_factor(fValue, fMomentum, fQuality, fVolStab, fFlow)

    momBonus = rsa > 2 ? 5 : rsa > 0.5 ? 2 : rsa < -1 ? -4 : 0
    riskAdjScore = f_clamp(math.round(score * riskMult + momBonus + (smc >= 60 ? 3 : 0)), 0, 100)
