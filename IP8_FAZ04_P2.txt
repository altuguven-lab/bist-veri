// Gunluk TL-hacim agirlikli yukselen/dusen gun orani (son 10 gun), log ile
// sikistirilip -2/+2 sinirlanmis. request.security() icinden cagrilacak.
// 14.09 DUZELTME A6 (kurul denetimi madde 5): fiyatin degismedigi (yatay)
// gunler eskiden "else" ile dusus hacmine yaziliyordu, hafif negatif
// yanlilik yaratiyordu. Artik yatay gunler tamamen disarida birakiliyor.
f_volFlowInner() =>
    upTurn = 0.0
    downTurn = 0.0
    for i = 1 to 10
        chg = close[i] - close[i+1]
        turn = close[i] * volume[i]
        if chg > 0
            upTurn += turn
        else if chg < 0
            downTurn += turn
    // 14.09 DUZELTME (kurul denetimi bolum 6): 10 gunun TAMAMI yataysa
    // (upTurn=0 VE downTurn=0), eski formul log(0.10)=-2.3 -> -2 (maksimum
    // NEGATIF) donduruyordu - tam yatay fiyat "en kotu para akisi" gibi
    // yanlis okunuyordu. Artik bu durumda NOTR (0.0) donuyor.
    upTurn == 0.0 and downTurn == 0.0 ? 0.0 : downTurn == 0.0 ? 2.0 : upTurn == 0.0 ? -2.0 : math.max(-2.0, math.min(2.0, math.log(upTurn / downTurn)))

f_fetch(sym) =>
    [c, r1, r5, r20, r60, pr5, v, rvol, ema20, vf] = request.security(sym, analizTF,
         [close[1],
          close[1]/close[2] - 1,
          close[1]/close[6] - 1,
          close[1]/close[21] - 1,
          close[1]/close[61] - 1,
          close[6]/close[11] - 1,
          volume[1],
          volume[1] / ta.sma(volume, 20)[1],
          ta.ema(close, 20)[1],
          f_volFlowInner()],
         lookahead = barmerge.lookahead_on, ignore_invalid_symbol = true)
    // 14.09 EKLENTI P0.4 (kurul denetimi - kritik, ortak gecerli-uye
    // maskesi): eskiden R5/R20/R60/PR5/VF BAGIMSIZ na-guvenli ortalaniyordu

    // FARKLI hisse kumelerini temsil edebiliyordu. Artik maske BURADA, tek
    // yerde uygulaniyor: hisse TUM 5 metrikte gecerliyse hepsi kalir,
    // degilse HEPSI birden na olur - tutarlilik otomatik saglanir.
    ok = not na(r5) and not na(r20) and not na(r60) and not na(pr5) and not na(vf)
    r5m = ok ? r5 : float(na)
    r20m = ok ? r20 : float(na)
    r60m = ok ? r60 : float(na)
    pr5m = ok ? pr5 : float(na)
    vfm = ok ? vf : float(na)
    [c, r1, r5m, r20m, r60m, pr5m, v, rvol, ema20, vfm]

f_naSafeAvg(arr) =>
    validSum = 0.0
    validCount = 0
    for v in arr
        if not na(v)
            validSum += v
            validCount += 1
    validCount >= 1 ? validSum / validCount : na

f_validCount(arr) =>
    cnt = 0
    for v in arr
        if not na(v)
            cnt += 1
    cnt

// ============================================================
// SRS - KESITSEL (CROSS-SECTIONAL) Z-SKOR
// Zaman-serisi (60 gunluk rolling) z-skoru yerine, AYNI barda 11 sektor
// ARASINDA karsilastirma yapiliyor - "bu sektor bugun digerlerine gore
// nasil" sorusunu cevaplar. Rolling ta.sma/ta.stdev grafik-bar sorununa
// hic girmeden guvenilir sekilde uygulanabiliyor. SRS 0-100, 50=notr.
// ============================================================
// 14.09 DUZELTME A4 (kurul denetimi madde 9 - CIDDI): eskiden array.avg()/
// array.stdev() dizide TEK bir na sektor varsa TUM sonucu na yapiyordu -
// bir sektorde veri eksikse butun 11 sektorun z-skoru/SRS'si bozulabiliyordu.
// Artik yalnizca GECERLI (na olmayan) degerler ortalama/std hesabina giriyor.
f_crossZ(arr, idx) =>
    sm = 0.0
    cnt = 0
    for v in arr
        if not na(v)
            sm += v
            cnt += 1
    ort = cnt > 0 ? sm / cnt : na
    sq = 0.0
    for v in arr
        if not na(v)
            sq += math.pow(v - ort, 2)
    std = cnt > 1 ? math.sqrt(sq / cnt) : na
    deger = array.get(arr, idx)
    // 14.09 DUZELTME P0.3 (kurul denetimi - kritik): std==0 durumu (butun
    // sektorlerin bu bilesende AYNI deger tasidigi) onceki surumde na
    // donduruyordu - bu VERI EKSIKLIGI degil, "bu bilesen bugun sektorleri
    // AYIRAMIYOR" anlamina gelir, katkisi NOTR (0.0) olmali, TUM SRS'yi
    // na yapmamali. Ayrica en az 8/11 sektorde gecerli veri sarti eklendi -
    // kesitsel karsilastirma cok az sektorle guvenilmez.
    // 14.09 EKLENTI P1 (kurul denetimi - "kolay ilk adim"): sadece 11
    // sektor oldugu icin klasik z-skor TEK bir uc deger tarafindan
    // bozulabilir (orn. bir sektor +%50 hareket ederse ortalama/std'yi
    // domine eder). Tam robust/winsorized z-skor yerine, basit ve ucuz bir
    // onlem olarak z-skor +-2.5 ile sinirlaniyor.
    hamZ = (na(deger) or cnt < 8) ? float(na) : (std <= 1e-10 ? 0.0 : (deger - ort) / std)
    na(hamZ) ? float(na) : math.max(-2.5, math.min(2.5, hamZ))

// FAZ 0.4 - ENTRY QUALITY SCORE (EQS) VE BIRLESIK AKSIYON
// SRS "sektor ne kadar guclu" sorusunu cevaplar - EQS "SU AN girmek ne
// kadar mantikli" sorusunu cevaplar. Ikisi BILEREK ayri: yuksek SRS +
// dusuk EQS = "gercek lider ama kovalanmaz" (orn. Rafineri +14.73% 5G).

// Asiri uzama/FOMO cezasi: sektorun 5 gunluk hareketi ne kadar sertse
// (asiri isinmis), giris kalitesi o kadar dusuruluyor.
// 14.09 DUZELTME P0.1 (kurul denetimi - KRITIK): eski formul "ret5pct<=5"
// kosulunu kullaniyordu - bu, +%3 ile -%30'u AYNI (100 puan) degerlendirdi.
// Sert dususler "uygun maliyetli giris" gibi puanlaniyordu (Demir-Celik'in
// dusuk SRS'ye ragmen GIR cikmasinin ana nedeni buydu). Artik hem asiri
// pozitif hem asiri negatif uzama cezalandiriliyor.
f_uzamaSkoru(ret5) =>
    p = ret5 * 100
    na(p) ? float(na) : p > 12 ? 0.0 : p > 5 ? 100.0 - (p - 5) * 8 : p < -12 ? 0.0 : p < -5 ? 100.0 - (math.abs(p) - 5) * 12 : 100.0

f_trendHizalamaSkoru(evre) =>
    evre == "ONCU" ? 100.0 : evre == "GUCLENIYOR" ? 75.0 : evre == "ZAYIFLIYOR" ? 40.0 : 10.0

f_rejimSkoru(rejim) =>
    rejim == "RISK-ON" ? 100.0 : rejim == "NOTR" ? 60.0 : rejim == "POZITIF SOK" ? 50.0 : rejim == "RISK-OFF" ? 25.0 : 0.0

f_liderKaliteSkoru(topV, kaliteliV) =>
    topV == "LIDER YOK" ? 0.0 : kaliteliV ? 100.0 : 50.0

f_eqs(srs, ret5, evre, topV, kaliteliV) =>
    na(srs) ? float(na) : 0.25*srs + 0.25*f_uzamaSkoru(ret5) + 0.20*f_trendHizalamaSkoru(evre) + 0.15*f_rejimSkoru(marketRejim) + 0.15*f_liderKaliteSkoru(topV, kaliteliV)

// 14.09 EKLENTI (denetim - EQS hala ham SRS ile hesaplaniyordu): SRS_adj'e
// paralel olarak EQS_adj eklendi. Karar mantigi (GIR/TEYIT BEKLE/KACIN)
// artik EQS_ADJ kullaniyor - ham EQS sadece GOSTERIM icin ayrica tutuluyor.
f_eqsAdj(eqs, gecerli) => na(eqs) ? float(na) : 50 + f_confKatsayi(gecerli) * (eqs - 50)

// Birlesik aksiyon: EQS + evre + asiri uzama + rejimi TEK etikette birlestirir.
// 14.09 DUZELTME P0.2/P0.5 (kurul denetimi - kritik): eskiden GIR icin
// sadece EQS>=65 ve evre yeterliydi - dusuk guvenli (2 uyeli) sektorler ve
// dusuk HAM SRS'ye sahip sektorler (Demir-Celik SRS 41,71 gibi) bile GIR
// alabiliyordu. Artik guven-ayarli SRS (srsAdj), Delta1 (zayiflamiyor mu)
// ve guven seviyesinin kendisi de GIR kapisina dahil.
// 14.09 DUZELTME (kurul denetimi madde D): eskiden EQS>=45 olan HEMEN HER
// durum "IZLE" oluyordu - dusuk kaliteli sektorler de (srsAdj<45) takip
// listesine giriyordu. Artik ayrim var: srsAdj<45 -> KACIN, digerleri ->
// TEYIT BEKLE (eski "IZLE"nin daha aciklayici adi).
// 14.09 DUZELTME (denetim): (1) "eqs" parametresi artik CAGRI NOKTASINDA
// EQS_ADJ olarak gecirilecek - karar mantigi guven-ayarli skorla calisiyor.
// (2) Delta3 teyidi eklendi - tek gunluk Delta1 gurultulu olabilir, GIR
// icin artik hem Delta1>=0 HEM Delta3>0 (3 gunluk egilim de pozitif)
// gerekiyor.
// 15.09 DUZELTME (denetim madde 8): "delta1 >= 0" gibi ifadeler Pine'da na
// ile karsilastirildiginda sessizce false donuyordu - COKME riski yoktu
// ama IMPLICIT (ortuk) davranisa guveniliyordu. Artik her kontrol ACIKCA
// "once na mi diye bak" seklinde yaziliyor - okunurluk ve gelecekte
// bakim yapacak kisi icin netlik amacli.
f_valid(x) => not na(x)

f_deltaOK(d1, d3) => f_valid(d1) and f_valid(d3) and d1 >= 0 and d3 > 0

    else if riskOff and dusukGuven

    else if dusukGuven

    else if eqsAdj >= 65 and f_valid(srsAdj) and srsAdj >= 52 and f_deltaOK(delta1, delta3) and trendOK
        sonuc := "GIR"
    else if f_valid(srsAdj) and srsAdj < 45
        sonuc := "KACIN"
    else if eqsAdj >= 45
        sonuc := "TEYIT BEKLE"
    sonuc

// LIDER HISSE (confirmed gunluk RS20 vs XU100 + RVOL + trend)
// Trend filtresini (c > EMA20) gecemeyen hicbir uye yoksa "LIDER YOK".
// ============================================================
// 14.09 EKLENTI D1 (Faz 0.3B madde 9 - standardize Leader Score): eskiden
// stockRS20*0.6 + (rvol-1)*20*0.4 kullaniliyordu - RS20 (~yuzde birkac) ile
// RVOL (0-5 araligi) FARKLI OLCEKTEYDI, RVOL sozde %40 agirliga ragmen
// skoru domine edebiliyordu. Artik her bilesen ONCE 0-100 olcegine tasinip
// SONRA agirliklandiriliyor. Ayrica hisse/SEKTOR RS20 eklendi (D2) - sadece
// XU100'e gore degil, kendi sektorune gore de goreli guc olcusu.
// 14.09 EKLENTI P1 (kurul denetimi - leave-one-out sektor lideri): eskiden
// her aday KENDI KATKISINI ICEREN sektor ortalamasina karsi olculuyordu -
// ozellikle 2 uyeli sektorlerde PETKM gibi bir hisse, kendi getirisiyle
// sisirdigi ortalamaya karsi kiyaslaniyordu (haksiz avantaj/dezavantaj).
// Artik her aday, KENDISI HARIC diger uyelerin ortalamasina gore olculuyor.
f_leaveOneOut(sektorRet, sektorGecerli, stockRet) =>
    (sektorGecerli <= 1 or na(stockRet)) ? float(na) : (sektorRet * sektorGecerli - stockRet) / (sektorGecerli - 1)

f_leaderRaw(stockR20, rvol, c, ema20, sektorR20) =>
    trendOK = c > ema20
    xuRS = ((1 + stockR20) / (1 + xuR20) - 1) * 100
    sekRS = ((1 + stockR20) / (1 + sektorR20) - 1) * 100
    rsScore = math.max(0.0, math.min(100.0, 50 + xuRS * 4))
    sekRsScore = math.max(0.0, math.min(100.0, 50 + sekRS * 4))
    rvolScore = math.max(0.0, math.min(100.0, 50 + (rvol - 1) * 25))
    trendOK ? sekRsScore * 0.4 + rsScore * 0.3 + rvolScore * 0.3 : float(na)

// 14.09 EKLENTI D3 (Faz 0.3B madde 7-8 - sektor birincisi/kaliteli lider
// ayrimi + minimum kalite esigi): artik ISIM ile birlikte, kazananin
// standardize skorunun kalite esigini (55) gecip gecmedigi de donduruluyor.
// Esigi gecmezse "sektor birincisi" olur ama "kaliteli/alinabilir lider"
// SAYILMAZ - ayri bir etiketle gosterilecek.
f_liderSec(isimArr, skorArr) =>
    enIyiIsim = "LIDER YOK"
    enIyiSkor = -1e10
    bulunduMu = false
    for i = 0 to array.size(isimArr) - 1
        s = array.get(skorArr, i)
        if not na(s) and (not bulunduMu or s > enIyiSkor)
            enIyiSkor := s
            enIyiIsim := array.get(isimArr, i)
            bulunduMu := true
    kaliteliMi = bulunduMu and enIyiSkor >= 55
    // 14.09 EKLENTI (kurul denetimi madde G): artik sadece isim degil,
    // liderin GERCEK SKORU da donduruluyor - ekranda "★PETKM -4.13%" gibi
    // 3 farkli kavrami (kalite/isim/gunluk getiri) karistiran gosterimi
    // duzeltebilmek icin.
    donenSkor = bulunduMu ? enIyiSkor : float(na)
    [enIyiIsim, donenSkor, kaliteliMi]
